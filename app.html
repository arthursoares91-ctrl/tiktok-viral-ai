🔧 Guia de Integração de Pagamentos
Stripe
1. Crie conta em stripe.com
2. Crie produtos/preços mensais
3. Configure webhook /api/webhook
4. Use Stripe Checkout ou Payment Links
5. Salve subscription_id no Supabase
Mercado Pago
1. Crie conta em mercadopago.com.br
2. Use API de Assinaturas (Subscriptions)
3. Configure PIX + Cartão
4. Webhook em /api/mp-webhook
5. Atualize plano no Supabase ao receber pagamento
Schema Supabase recomendado
-- profiles (extende auth.users)
CREATE TABLE profiles (
  id UUID REFERENCES auth.users PRIMARY KEY,
  name TEXT,
  plan TEXT DEFAULT 'free',       -- free | pro | premium
  credits_today INT DEFAULT 5,
  credits_reset_at TIMESTAMPTZ,
  stripe_customer_id TEXT,
  mp_subscription_id TEXT,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- generations (histórico)
CREATE TABLE generations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES profiles(id) ON DELETE CASCADE,
  type TEXT,                      -- ideias|roteiro|legenda|hashtags|analisador|serie
  input JSONB,
  output TEXT,
  created_at TIMESTAMPTZ DEFAULT now()
);
